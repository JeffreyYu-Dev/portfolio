# Yail Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the Mail window with Yail, a Gmail recreation served at the fake domain `yail.com` inside the Zen browser. It has an inbox with Primary/Promotions/Social tabs, stars, search, an email view, and Gmail's floating compose that really sends to Jeffrey.

**Architecture:** Yail is a set of TanStack routes under `/yail`. The browser's address helpers map them to `yail.com`. Everything that can break quietly is a pure, unit-tested module:
- address mapping in `browser-address.ts`
- mailbox lists, search and titles in `yail.ts`
- stars and draft discard in `mail-store.ts`
- compose and toast state in two small stores

The route files and `src/components/yail/*` only render those. The sending pipeline (schema, server function, spam checks, Resend) is reused unchanged.

**Tech Stack:** TanStack Start 1.168 / Router 1.170 (file routes, `validateSearch`, `redirect`), React 19, Tailwind v4, lucide-react, Zod 4, Vitest 5, npm.

**Spec:** `docs/superpowers/specs/2026-10-13-yail-design.md` (and, for sending, `docs/superpowers/specs/2026-10-13-mail-app-design.md`)

## Review Focus

1. **Deep link to the visitor's own sent email** (`yail.com/sent/<id>`). The server render has an empty store, so it must not flash "This message doesn't exist" before saved state loads. Guarded with `useMailReady` in Task 6, with a manual check in Task 9.
2. **Minimising compose and restoring it** must keep text typed in the last 500ms. Compose stays mounted (hidden) while minimised (Task 7), with a manual check.
3. **Discarding while a draft save is pending** must not bring the draft back. `discardDraft()` cancels the pending save, pinned by a test in Task 2.
4. **Numeric or odd search queries** (`?q=2026`, which TanStack parses to a number, or `?q=`). `searchQuery()` normalises the raw value and a blank query shows no results, pinned by tests in Task 3.
5. **Typed addresses for Yail in every shape** (`YAIL.com`, `www.yail.com/starred/`, `https://yail.com/search?q=x`, `yail.com?compose=1`), pinned by tests in Task 1.

import { type Email, inbox, type MailTab } from "#/data/mail";
import type { MailState, SentEmail } from "#/lib/mail-store";

// what yail shows: the seeded inbox plus the visitor's own sent mail and draft,
// sorted and filtered the way gmail does. pure, the routes just render it
export type Box = "inbox" | "starred" | "sent" | "drafts";
export const BOXES: Box[] = ["inbox", "starred", "sent", "drafts"];
export const TABS: MailTab[] = ["primary", "promotions", "social"];

export type Item = {
  id: string;
  box: "inbox" | "sent" | "drafts";
  from: string;
  fromAddress: string;
  subject: string;
  date: string;
  body: string;
  tab?: MailTab;
  unread: boolean;
  starred: boolean;
};

const TITLES: Record<Box, string> = {
  inbox: "Inbox",
  starred: "Starred",
  sent: "Sent",
  drafts: "Drafts",
};

export const isBox = (s: unknown): s is Box => BOXES.includes(s as Box);
export const isTab = (s: unknown): s is MailTab => TABS.includes(s as MailTab);
export const boxTitle = (box: Box) => TITLES[box];

const newestFirst = (a: Item, b: Item) => Date.parse(b.date) - Date.parse(a.date);

function fromInbox(email: Email, state: MailState): Item {
  return {
    ...email,
    box: "inbox",
    unread: !state.read.includes(email.id),
    starred: state.starred.includes(email.id),
  };
}

function fromSent(email: SentEmail, state: MailState): Item {
  return {
    ...email,
    box: "sent",
    from: "To: Jeffrey Yu",
    fromAddress: "",
    unread: false,
    starred: state.starred.includes(email.id),
  };
}

function everything(state: MailState, emails: Email[]) {
  return [
    ...emails.map((e) => fromInbox(e, state)),
    ...state.sent.map((e) => fromSent(e, state)),
  ];
}

export function listFor(
  box: Box,
  tab: MailTab,
  state: MailState,
  emails: Email[] = inbox,
): Item[] {
  if (box === "drafts") {
    const draft = state.draft;
    if (!draft) return [];
    return [
      {
        id: "draft",
        box: "drafts",
        from: "Draft",
        fromAddress: "",
        subject: draft.subject.trim() || "(no subject)",
        date: new Date(draft.startedAt).toISOString(),
        body: draft.body,
        unread: false,
        starred: false,
      },
    ];
  }
  const all = everything(state, emails);
  const items =
    box === "inbox"
      ? all.filter((i) => i.box === "inbox" && i.tab === tab)
      : box === "sent"
        ? all.filter((i) => i.box === "sent")
        : all.filter((i) => i.starred);
  return items.sort(newestFirst);
}

export function findItem(
  box: Box,
  id: string,
  state: MailState,
  emails: Email[] = inbox,
): Item | null {
  if (box === "drafts") return null;
  const all = everything(state, emails);
  const item = all.find((i) => i.id === id);
  if (!item) return null;
  if (box === "starred") return item.starred ? item : null;
  return item.box === box ? item : null;
}

export function search(
  query: string,
  state: MailState,
  emails: Email[] = inbox,
): Item[] {
  const q = query.trim().toLowerCase();
  if (!q) return [];
  return everything(state, emails)
    .filter((i) =>
      [i.from, i.subject, i.body].some((text) => text.toLowerCase().includes(q)),
    )
    .sort(newestFirst);
}

export function unreadByTab(
  state: MailState,
  emails: Email[] = inbox,
): Record<MailTab, number> {
  const counts = { primary: 0, promotions: 0, social: 0 };
  for (const email of emails)
    if (!state.read.includes(email.id)) counts[email.tab]++;
  return counts;
}

// the router parses ?q=2026 into a number, and a hand-typed url can be anything
export function searchQuery(raw: unknown): string {
  if (typeof raw === "string") return raw;
  if (typeof raw === "number") return String(raw);
  return "";
}

// tab titles like gmail's. pure, so it only knows seeded subjects, a sent
// email's tab says "Sent"
export function yailTitle(path: string, emails: Email[] = inbox): string {
  const [, , box, id] = path.split("/");
  if (box === "search") return "Search results – Yail";
  if (!isBox(box)) return "Yail";
  const seeded =
    id && (box === "inbox" || box === "starred")
      ? emails.find((e) => e.id === id)
      : undefined;
  return `${seeded ? seeded.subject : boxTitle(box)} – Yail`;
}

import { useSyncExternalStore } from "react";
import type { Draft } from "#/lib/mail-store";

// gmail's floating new message window. module state, so it stays open while the
// visitor moves between yail pages. the text itself lives in mail-store's draft
export type ComposeMode = "closed" | "open" | "minimised" | "expanded";
type ComposeState = { mode: ComposeMode; initial: Draft | null; key: number };

const closed: ComposeState = { mode: "closed", initial: null, key: 0 };
let state = closed;
const listeners = new Set<() => void>();

function set(next: ComposeState) {
  state = next;
  for (const listener of listeners) listener();
}

export const getCompose = () => state;

// a new key remounts the form, so a reply opened over another message starts fresh
export function openCompose(draft: Draft) {
  set({ mode: "open", initial: draft, key: state.key + 1 });
}

function mode(next: ComposeMode) {
  if (state.mode !== "closed") set({ ...state, mode: next });
}

export const minimise = () => mode("minimised");
export const expand = () => mode("expanded");
export const restore = () => mode("open");

export function closeCompose() {
  set({ ...closed, key: state.key });
}

export function useCompose() {
  return useSyncExternalStore(
    (listener) => {
      listeners.add(listener);
      return () => {
        listeners.delete(listener);
      };
    },
    () => state,
    () => closed,
  );
}

// gmail's "message sent" box in the corner. it goes away on its own
export const TOAST_MS = 6000;

type ToastState = { sentId: string | null };
const hidden: ToastState = { sentId: null };
let state = hidden;
let timer: ReturnType<typeof setTimeout> | undefined;
const listeners = new Set<() => void>();

function set(next: ToastState) {
  state = next;
  for (const listener of listeners) listener();
}

export const getToast = () => state;

export function showSent(id: string) {
  clearTimeout(timer);
  set({ sentId: id });
  timer = setTimeout(dismiss, TOAST_MS);
}

export function dismiss() {
  clearTimeout(timer);
  set(hidden);
}

export function useToast() {
  return useSyncExternalStore(
    (listener) => {
      listeners.add(listener);
      return () => {
        listeners.delete(listener);
      };
    },
    () => state,
    () => hidden,
  );
}

// the fields of a new message to jeffrey. the draft is saved as they type, so
// closing, minimising or a failed send never loses what they wrote
export function Compose({
  initial,
  onSent,
  onDiscard,
}: {
  initial: Draft;
  onSent: (sent: SentEmail) => void;
  onDiscard: () => void;
}) {
  const [fields, setFields] = useState<MailFields>({
    name: initial.name,
    email: initial.email,
    subject: initial.subject,
    body: initial.body,
  });
  const [errors, setErrors] = useState<FieldErrors>({});
  const [sending, setSending] = useState(false);
  const [failed, setFailed] = useState(false);
  const [confirming, setConfirming] = useState(false);
  const honeypot = useRef<HTMLInputElement>(null);
  const id = useId();

  // closing mid-sentence still keeps the last few keystrokes
  useEffect(() => flushDraft, []);

  const change = (key: keyof MailFields) => (value: string) => {
    const next = { ...fields, [key]: value };
    setFields(next);
    setErrors((e) => ({ ...e, [key]: undefined }));
    saveDraft({ ...next, startedAt: initial.startedAt });
  };

  function discard() {
    const written = Object.values(fields).some((v) => v.trim());
    if (written && !confirming) {
      setConfirming(true);
      return;
    }
    onDiscard();
  }

  async function send(e: FormEvent) {
    e.preventDefault();
    const checked = checkMail(fields);
    if (!checked.ok) {
      setErrors(checked.errors);
      return;
    }
    setSending(true);
    setFailed(false);
    try {
      const result = await sendMail({
        data: {
          ...fields,
          website: honeypot.current?.value ?? "",
          filledMs: Date.now() - initial.startedAt,

      });
      if (!result.ok) {
        setFailed(true);
        return;
      }
      const sent = {
        id: crypto.randomUUID(),
        subject: checked.mail.subject,
        body: checked.mail.body,
        date: new Date().toISOString(),
      };
      recordSent(sent);
      onSent(sent);
    } catch {
      setFailed(true);
    } finally {
      setSending(false);
    }
  }

  const line = "flex items-center gap-2 border-b border-[#e0e0e0] py-2 text-sm";
  const input = "min-w-0 flex-1 bg-transparent outline-none placeholder:text-[#5f6368]";
